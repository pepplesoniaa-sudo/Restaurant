POWERSHELL LOOP FOR YOU TO SCREENSHOT (the same loop I ran; run it yourself for a screenshot):
  $url = "https://restaurant-api-iota-eight.vercel.app/api/v1/restaurants?limit=1"
  for ($i = 1; $i -le 110; $i++) {
    $r = curl.exe -s -o NUL -w "%{http_code}" $url
    Write-Output ("req {0}: {1}" -f $i, $r)
    if ($r -eq "429") {
      Write-Output "--- 429 headers and body ---"
      curl.exe -s -i $url
      break
    }
  }

  window, run several copies at once, or send the requests as fast as possible; 110 sequential
