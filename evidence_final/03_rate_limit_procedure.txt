Step 1. Burst (100 allowed -> 200, then the rest -> 429). Paste and run:

  $url = "https://restaurant-api-iota-eight.vercel.app/api/v1/restaurants?limit=1"
  $n = 200
  $jobs = 1..$n | ForEach-Object {
    [powershell]::Create().AddScript({
      param($u) try {
        (Invoke-WebRequest -Uri $u -UseBasicParsing -TimeoutSec 30).StatusCode
      } catch { [int]$_.Exception.Response.StatusCode }
    }).AddArgument($url)
  }
  # OR use the same runspace approach previously used (100x200 + 100x429).
  # Run them and tally the status codes:
  $jobs | ForEach-Object { $_.BeginInvoke() | Out-Null }
  Start-Sleep -Seconds 5
  $results = $jobs | ForEach-Object { $_.EndInvoke($_.BeginInvoke()) }  # adapt to your script
  $results | Group-Object | Select-Object Count, Name

  Expect: one group "200" and one group "429" (counts summing to 200).
